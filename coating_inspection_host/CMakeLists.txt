cmake_minimum_required(VERSION 3.18)
project(coating_inspection VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(INSPECTION_SANITIZE "启用地址与未定义行为检查" OFF)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
find_path(MODBUS_INCLUDE_DIR modbus/modbus.h REQUIRED)
find_library(MODBUS_LIBRARY modbus REQUIRED)
add_library(thickness_protocol vendor/thickness/src/protocol.cpp)
target_include_directories(thickness_protocol PUBLIC vendor/thickness/include)
add_library(thickness_bluez vendor/thickness/src/bluez_reader.cpp)
target_link_libraries(thickness_bluez PUBLIC thickness_protocol PRIVATE PkgConfig::GIO)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/config/migrations/001_init.sql SCHEMA_SQL)
configure_file(include/inspection/schema.hpp.in generated/inspection/schema.hpp @ONLY)
add_library(inspection
  src/support/config.cpp src/support/runtime.cpp
  src/instrument/instrument.cpp src/workstation/protocol.cpp src/workstation/workstation.cpp
  src/persistence/repository.cpp src/cloud/cloud.cpp src/cli/cli.cpp src/cli/commands.cpp src/domain/worker.cpp)
target_include_directories(inspection PUBLIC include PRIVATE ${MODBUS_INCLUDE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(inspection PUBLIC Threads::Threads SQLite::SQLite3 yaml-cpp nlohmann_json::nlohmann_json thickness_bluez PRIVATE ${MODBUS_LIBRARY} PkgConfig::GIO)
target_compile_options(inspection PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(INSPECTION_SANITIZE)
  target_compile_options(thickness_protocol PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(thickness_bluez PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(inspection PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(inspection PUBLIC -fsanitize=address,undefined)
endif()
add_executable(inspection_service app/main.cpp)
target_link_libraries(inspection_service PRIVATE inspection)
include(CTest)
if(BUILD_TESTING)
  add_executable(inspection_tests tests/unit.cpp)
  target_link_libraries(inspection_tests PRIVATE inspection)
  add_test(NAME unit COMMAND inspection_tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME simulator_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/simulator_contract.py)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:inspection_service>)
  set_tests_properties(integration PROPERTIES TIMEOUT 180)
endif()
install(TARGETS inspection_service RUNTIME DESTINATION bin)
install(DIRECTORY config/ DESTINATION share/coating-inspection/config)
